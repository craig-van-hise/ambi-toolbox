```typescript
import { spawn } from 'child_process';
import path from 'path';

ipcMain.handle('process-ambi-rotate', async (event, filePaths: string[], rotation) => {
    console.log("MAIN: Starting Rotation Render...", { filePaths, rotation });

    // 1. Locate the Python Script
    // Adjust 'ambi_rotate.py' to whatever your actual script name is. 
    // If you aren't sure, check the 'py' or 'scripts' folder.
    const scriptPath = path.join(__dirname, '../../py/ambi_rotate.py'); // Adjust path as needed for dev/prod

    return new Promise((resolve, reject) => {
        // 2. Spawn Python Process
        // We pass the files and rotation args as JSON or flags
        const python = spawn('python3', [
            scriptPath,
            '--files', JSON.stringify(filePaths),
            '--yaw', rotation.yaw.toString(),
            '--pitch', rotation.pitch.toString(),
            '--roll', rotation.roll.toString()
        ]);

        let output = '';
        let errorOutput = '';

        python.stdout.on('data', (data) => {
            console.log(`PYTHON LOG: ${data}`);
            output += data.toString();
        });

        python.stderr.on('data', (data) => {
            console.error(`PYTHON ERR: ${data}`);
            errorOutput += data.toString();
        });

        python.on('close', (code) => {
            if (code === 0) {
                console.log("MAIN: Rotation Render Success.");
                resolve({ success: true, log: output });
            } else {
                console.error("MAIN: Rotation Render Failed.", code);
                reject(new Error(`Python script failed with code ${code}: ${errorOutput}`));
            }
        });
    });
});

```
