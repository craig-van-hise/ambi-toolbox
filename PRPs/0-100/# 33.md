```typescript
// Inside PreviewEngine class

play() {
    if (!this.sourceNode || this.isPlaying) return;

    // 1. Ensure the Gain starts at SILENCE to prevent clicks
    // Cancel any scheduled changes to be safe
    this.gainNode.gain.cancelScheduledValues(this.audioContext.currentTime);
    this.gainNode.gain.setValueAtTime(0, this.audioContext.currentTime);

    // 2. Start the Source
    // Note: If you need to recreate the source node (because they are one-time use), do it here.
    // ... (Your existing source creation logic) ...
    
    this.sourceNode.start(0, this.currentOffset);

    // 3. The "Anti-Pop" Micro-Fade (20ms)
    // Ramp volume from 0 to 1 over 0.02 seconds
    this.gainNode.gain.linearRampToValueAtTime(1.0, this.audioContext.currentTime + 0.02);

    this.isPlaying = true;
    this.startedAt = this.audioContext.currentTime - this.pausedAt;
}

pause() {
    if (!this.sourceNode || !this.isPlaying) return;

    // Optional: Micro-fade OUT to prevent "Stop Click" too
    const fadeOutTime = 0.01; // 10ms
    this.gainNode.gain.cancelScheduledValues(this.audioContext.currentTime);
    this.gainNode.gain.setValueAtTime(this.gainNode.gain.value, this.audioContext.currentTime);
    this.gainNode.gain.linearRampToValueAtTime(0, this.audioContext.currentTime + fadeOutTime);

    // Stop after fade
    setTimeout(() => {
        if (this.sourceNode) {
            this.sourceNode.stop();
            this.pausedAt = this.getCurrentTime();
            this.isPlaying = false;
        }
    }, fadeOutTime * 1000);
}

```
