```typescript
export class NativeRotator {
    private context: AudioContext;
    private splitter: ChannelSplitterNode;
    private merger: ChannelMergerNode;
    private yawGainXX: GainNode;
    private yawGainXY: GainNode;
    private yawGainYX: GainNode;
    private yawGainYY: GainNode;
    
    // Gain nodes for the Stereo Decode (Simple Cardioids)
    private outputL: GainNode;
    private outputR: GainNode;

    constructor(ctx: AudioContext) {
        this.context = ctx;
        
        // 1. Input Node (We expect 4 channels: W, Y, Z, X)
        this.splitter = ctx.createChannelSplitter(4);
        
        // 2. The Matrix Mixer for Rotation
        // Formula:
        // X_new = X*cos(a) - Y*sin(a)
        // Y_new = X*sin(a) + Y*cos(a)
        
        this.yawGainXX = ctx.createGain(); // X -> X (cos)
        this.yawGainXY = ctx.createGain(); // Y -> X (-sin)
        this.yawGainYX = ctx.createGain(); // X -> Y (sin)
        this.yawGainYY = ctx.createGain(); // Y -> Y (cos)
        
        // 3. Stereo Decoder (Monitor Output)
        // L = W + 0.5*Y_new
        // R = W - 0.5*Y_new
        this.merger = ctx.createChannelMerger(2);
        
        this.outputL = ctx.createGain();
        this.outputR = ctx.createGain();
        
        // Wiring
        // Channel Map: 0=W, 1=Y, 2=Z, 3=X (ACN Standard)
        
        // Rotation Wiring
        // X (Ch 3) flows to...
        this.splitter.connect(this.yawGainXX, 3);
        this.splitter.connect(this.yawGainYX, 3);
        
        // Y (Ch 1) flows to...
        this.splitter.connect(this.yawGainXY, 1);
        this.splitter.connect(this.yawGainYY, 1);
        
        // Decoding Wiring (Simple Stereo Monitor)
        // W (Ch 0) -> Left & Right
        this.splitter.connect(this.merger, 0, 0); // W to L
        this.splitter.connect(this.merger, 0, 1); // W to R
        
        // Rotated Y components sum to create Y_new, then to L/R
        // We sum them at the merger via intermediate gains if needed, 
        // but for simplicity we can connect gains directly to merger with scaling.
        
        // L = W + Y_new (Left is positive Y)
        this.yawGainYX.connect(this.merger, 0, 0); // X contribution to Y -> L
        this.yawGainYY.connect(this.merger, 0, 0); // Y contribution to Y -> L
        
        // R = W - Y_new (Right is negative Y)
        // We need inverters for the Right channel.
        const inverter = ctx.createGain();
        inverter.gain.value = -1.0;
        
        this.yawGainYX.connect(inverter);
        this.yawGainYY.connect(inverter);
        inverter.connect(this.merger, 0, 1); // Inverted Y -> R
    }

    // The Input Node to connect your Source to
    get input() { return this.splitter; }
    
    // The Output Node to connect to Speakers
    get output() { return this.merger; }

    setRotation(yawDeg: number) {
        // Convert to Radians
        const rad = yawDeg * (Math.PI / 180);
        const c = Math.cos(rad);
        const s = Math.sin(rad);
        
        // Apply Matrix Coefficients
        // X' = X*c - Y*s
        this.yawGainXX.gain.value = c;
        this.yawGainXY.gain.value = -s;
        
        // Y' = X*s + Y*c
        this.yawGainYX.gain.value = s;
        this.yawGainYY.gain.value = c;
    }
}

```
