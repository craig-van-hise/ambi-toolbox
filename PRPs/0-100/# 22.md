```typescript
// ... inside setupAudio(fullBuffer) ...

// 1. Create a 4-Channel Proxy Buffer (1st Order)
// We ignore channels 4-15 for the preview to prevent the library crash.
console.log(`Creating 1st Order Proxy from ${fullBuffer.numberOfChannels}ch input...`);

const proxyBuffer = audioContext.createBuffer(4, fullBuffer.length, fullBuffer.sampleRate);

// Copy the first 4 channels (W, Y, Z, X)
// Note: We assume ACN ordering. If input is FuMa, 0-3 is still WXYZ, so it works for monitoring.
for (let i = 0; i < 4; i++) {
    // Safety check: ensure input actually has this channel
    if (i < fullBuffer.numberOfChannels) {
        proxyBuffer.copyToChannel(fullBuffer.getChannelData(i), i);
    }
}

// 2. Initialize the Library with Order = 1
const safeOrder = 1;
const Lib = window.ambisonics || window.JSAmbisonics || (window as any).ambisonics;

if (!Lib) throw new Error("JSAmbisonics library missing");

try {
    console.log("Initializing 1st Order Rotator...");
    // Force Order 1 to match our Proxy Buffer
    const rotator = new Lib.SceneRotator(safeOrder);
    
    console.log("Initializing 1st Order Binaural Decoder...");
    const decoder = new Lib.BinauralDecoder(audioContext, safeOrder);
    
    // 3. Create Source and Connect Graph
    // Important: We play the PROXY buffer, not the full buffer
    const source = audioContext.createBufferSource();
    source.buffer = proxyBuffer; 
    source.loop = this.isLooping || false;

    // Graph: Source(4ch) -> Rotator(4ch) -> Decoder(2ch) -> Gain -> Out
    source.connect(rotator.in);
    rotator.out.connect(decoder.in);
    decoder.out.connect(this.gainNode);
    this.gainNode.connect(audioContext.destination);

    // 4. Save References for the UI
    this.sourceNode = source;
    this.rotatorNode = rotator;

    console.log("✅ Spatial Preview Active (1st Order Proxy)");

} catch (e) {
    console.error("❌ 1st Order Proxy Failed:", e);
    // Fallback to Mono Pass-Through if even 1st Order fails
    this.setupPassThrough(proxyBuffer); 
}

```
