// Inside PreviewEngine class

play() {
    if (this.isPlaying) return;

    // 1. SAFETY: Ensure we start from SILENCE
    // This kills any residual buffer noise or non-zero crossing clicks.
    this.gainNode.gain.cancelScheduledValues(this.audioContext.currentTime);
    this.gainNode.gain.setValueAtTime(0, this.audioContext.currentTime);

    // 2. SOURCE MANAGEMENT
    // Recreate the source (AudioBufferSourceNodes are one-time use)
    this.sourceNode = this.audioContext.createBufferSource();
    this.sourceNode.buffer = this.fullBuffer || this.proxyBuffer; // Use the correct buffer
    this.sourceNode.loop = this.isLooping;
    if (this.isLooping) {
        this.sourceNode.loopStart = this.loopIn;
        this.sourceNode.loopEnd = this.loopOut;
    }

    // 3. RECONNECT GRAPH
    // (Ensure you connect to your Rotator input here)
    if (this.nativeRotator) {
        this.sourceNode.connect(this.nativeRotator.input);
    }

    // 4. DETERMINE START TIME
    // CRITICAL FIX: Use currentOffset (set by scrubber) if available
    const startTime = this.currentOffset;

    // 5. START PLAYBACK
    this.sourceNode.start(0, startTime);
    
    // 6. FADE IN (20ms) - The "Anti-Pop"
    this.gainNode.gain.linearRampToValueAtTime(1.0, this.audioContext.currentTime + 0.02);

    // 7. SYNC CLOCKS
    this.startedAt = this.audioContext.currentTime - startTime;
    this.pausedAt = startTime; // Sync pausedAt so pause logic holds up
    this.isPlaying = true;
}

stop() {
    if (this.sourceNode) {
        // Fast Fade Out (10ms) to prevent "Stop Click"
        try {
            this.gainNode.gain.cancelScheduledValues(this.audioContext.currentTime);
            this.gainNode.gain.setValueAtTime(this.gainNode.gain.value, this.audioContext.currentTime);
            this.gainNode.gain.linearRampToValueAtTime(0, this.audioContext.currentTime + 0.01);
            
            this.sourceNode.stop(this.audioContext.currentTime + 0.01);
        } catch (e) {
            // Ignore if already stopped
        }
    }
    this.isPlaying = false;
    this.currentOffset = 0; // Reset to beginning
    this.pausedAt = 0;
}

pause() {
    if (!this.isPlaying) return;
    
    // Calculate where we stopped so we can resume or seek from here
    const elapsed = this.audioContext.currentTime - this.startedAt;
    this.currentOffset = elapsed % this.duration;
    this.pausedAt = this.currentOffset;

    if (this.sourceNode) {
        // Fade Out (10ms)
        this.gainNode.gain.cancelScheduledValues(this.audioContext.currentTime);
        this.gainNode.gain.setValueAtTime(1, this.audioContext.currentTime);
        this.gainNode.gain.linearRampToValueAtTime(0, this.audioContext.currentTime + 0.01);
        this.sourceNode.stop(this.audioContext.currentTime + 0.01);
    }
    this.isPlaying = false;
}

seek(time: number) {
    // Clamp time
    time = Math.max(0, Math.min(time, this.duration));
    
    this.currentOffset = time;
    this.pausedAt = time; // IMPORTANT: Update pausedAt so Play() picks it up

    if (this.isPlaying) {
        // If playing, we must restart the stream at the new time
        this.sourceNode.stop(); // Stop old stream
        this.play(); // Restart immediately at new currentOffset
    }
}
