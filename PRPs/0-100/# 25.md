// NATIVE ARBITRARY-ORDER ROTATOR (Z-Axis / Yaw Only)
// This implementation strictly follows ACN channel ordering and SN3D normalization.
// It supports ANY order (1st, 3rd, 7th, 50th...) by applying sectorial rotation 
// to the corresponding (m, -m) harmonic pairs.

export class NativeRotator {
    private ctx: AudioContext;
    private splitter: ChannelSplitterNode;
    private merger: ChannelMergerNode;
    private gains: GainNode[] = []; // We keep refs to update values
    private channelCount: number;

    // Output Monitoring (Downmix to Stereo for Preview)
    private monitorMerger: ChannelMergerNode;
    public output: GainNode;

    constructor(ctx: AudioContext, channelCount: number) {
        this.ctx = ctx;
        this.channelCount = channelCount;

        // 1. Input/Output Nodes
        this.splitter = ctx.createChannelSplitter(channelCount);
        this.merger = ctx.createChannelMerger(channelCount);
        
        // Monitor Path (Simple Stereo Downmix of W+Y+X for orientation check)
        this.monitorMerger = ctx.createChannelMerger(2);
        this.output = ctx.createGain();

        // 2. Build the Arbitrary Matrix
        this.buildGraph();
    }

    private buildGraph() {
        // Iterate through every channel (ACN index)
        // ACN index 'i' maps to degree 'l' and order 'm' via:
        // l = floor(sqrt(i))
        // m = i - l^2 - l
        
        for (let i = 0; i < this.channelCount; i++) {
            const l = Math.floor(Math.sqrt(i));
            const m = i - (l * l) - l;

            // CASE A: Zonal Harmonics (m=0) -> Invariant
            // Examples: Ch 0 (W), Ch 2 (Z), Ch 6 (R), etc.
            // These DO NOT change during Yaw rotation. Pass through directly.
            if (m === 0) {
                const gain = this.ctx.createGain();
                gain.gain.value = 1.0;
                this.splitter.connect(gain, i);
                gain.connect(this.merger, 0, i);
            }

            // CASE B: Sectorial Pairs (m != 0)
            // We only process positive 'm'. We find its negative partner '-m'.
            // The pair is: ACN(l, m) and ACN(l, -m).
            else if (m > 0) {
                // Calculate partner index
                // Partner has same 'l' but order '-m'.
                const partnerIdx = (l * l) + l - m;
                const myIdx = i;

                // Create the 2x2 Matrix for this pair
                // [ Val_pos ]   [ cos(ma)  -sin(ma) ] [ In_pos ]
                // [ Val_neg ] = [ sin(ma)   cos(ma) ] [ In_neg ]
                
                // We need 4 Gains for this pair
                // 1. My -> My (cos)
                const g_mm = this.ctx.createGain();
                this.splitter.connect(g_mm, myIdx);
                g_mm.connect(this.merger, 0, myIdx);
                
                // 2. Partner -> My (sin)
                const g_pm = this.ctx.createGain();
                this.splitter.connect(g_pm, partnerIdx);
                g_pm.connect(this.merger, 0, myIdx);
                
                // 3. My -> Partner (-sin)
                const g_mp = this.ctx.createGain();
                this.splitter.connect(g_mp, myIdx);
                g_mp.connect(this.merger, 0, partnerIdx);
                
                // 4. Partner -> Partner (cos)
                const g_pp = this.ctx.createGain();
                this.splitter.connect(g_pp, partnerIdx);
                g_pp.connect(this.merger, 0, partnerIdx);
                
                // Store refs with metadata to update them later
                // We store 'm' so we know how fast to rotate this pair
                (g_mm as any)._rotType = 'cos'; (g_mm as any)._m = m;
                (g_pm as any)._rotType = 'sin'; (g_pm as any)._m = m;
                (g_mp as any)._rotType = 'nsin'; (g_mp as any)._m = m;
                (g_pp as any)._rotType = 'cos'; (g_pp as any)._m = m;

                this.gains.push(g_mm, g_pm, g_mp, g_pp);
            }
        }
        
        // 3. Setup Monitor (Downmix 1st Order to Stereo)
        // We take the ROTATED output from the merger
        // L = W + Y
        // R = W - Y
        // (We ignore X for simple L/R panning, or mix it in for "front/back" volume dip)
        
        // W (Ch 0)
        this.merger.connect(this.monitorMerger, 0, 0); // W -> L
        this.merger.connect(this.monitorMerger, 0, 1); // W -> R
        
        // Y (Ch 1)
        this.merger.connect(this.monitorMerger, 1, 0); // Y -> L (Left is +Y)
        
        const inv = this.ctx.createGain();
        inv.gain.value = -1.0;
        this.merger.connect(inv, 1, 0);
        inv.connect(this.monitorMerger, 0, 1); // -Y -> R
        
        this.monitorMerger.connect(this.output);
    }

    get input() { return this.splitter; }

    setRotation(yawDeg: number) {
        const rad = yawDeg * (Math.PI / 180);

        // Update all dynamic gains
        this.gains.forEach((g: any) => {
            const m = g._m; // The order multiplier
            const angle = rad * m; // Higher orders rotate faster!
            
            if (g._rotType === 'cos') g.gain.value = Math.cos(angle);
            if (g._rotType === 'sin') g.gain.value = Math.sin(angle);
            if (g._rotType === 'nsin') g.gain.value = -Math.sin(angle);
        });
    }
}
