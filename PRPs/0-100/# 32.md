```tsx
// Sync UI with Audio Engine
useEffect(() => {
    let animationFrameId: number;

    const tick = () => {
        if (previewEngine.isPlaying) {
            // 1. Get exact time from engine
            const time = previewEngine.getCurrentTime();
            
            // 2. Update State (Force Re-render)
            setCurrentTime(time);
            
            // 3. Check for Loop (Visual Snap)
            // If the audio engine loops, the time will jump back automatically.
            // We just need to keep polling.
        }
        animationFrameId = requestAnimationFrame(tick);
    };

    // Start loop
    tick();

    // Cleanup
    return () => cancelAnimationFrame(animationFrameId);
}, [previewEngine, isPlaying]); // Re-run if play state changes
```

```typescript
getCurrentTime() {
    if (!this.isPlaying) return this.pausedAt;
    // Calculate: (Current Audio Ctx Time - Start Time) % Loop Duration
    let rawTime = this.pausedAt + (this.audioContext.currentTime - this.startedAt);

    // Handle Loop Math for UI
    if (this.isLooping && this.loopOut > 0) {
         const loopDuration = this.loopOut - this.loopIn;
         if (rawTime >= this.loopOut) {
             // This is a naive visual fix; the AudioNode handles the real loop.
             // We just clamp visual for smoothness or wrap it?
             // Let's just return the raw time bounded by loop
             rawTime = this.loopIn + ((rawTime - this.loopIn) % loopDuration);
         }
    }
    return Math.min(rawTime, this.duration);
}
```
