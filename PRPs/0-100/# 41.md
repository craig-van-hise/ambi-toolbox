Replace `src/components/SmartDropZone.tsx` with this version. It supports **both** the new features (folders/browsing) and the old tools (legacy `onDrop`).

```tsx
import React, { useState } from 'react';

interface Props {
    // New Props (AmbiRotate)
    onFilesLoaded?: (files: { path: string, name: string }[]) => void;
    acceptedExtensions?: string[]; // e.g., ['wav', 'opus']
    
    // Legacy Props (Backwards Compatibility)
    onDrop?: (files: File[]) => void; 
    
    children?: React.ReactNode;
    className?: string;
}

export const SmartDropZone: React.FC<Props> = ({ 
    onFilesLoaded, 
    onDrop, 
    acceptedExtensions = ['wav', 'opus', 'caf', 'flac', 'mp3', 'ogg', 'aac'],
    children, 
    className 
}) => {
    const [isDragOver, setIsDragOver] = useState(false);

    // 1. HANDLER: Drop
    const handleDrop = async (e: React.DragEvent) => {
        e.preventDefault();
        setIsDragOver(false);
        
        // A. Legacy Path (Old Tools)
        // If the tool expects raw File objects, give them raw File objects.
        if (onDrop) {
            onDrop(Array.from(e.dataTransfer.files));
            return; 
        }

        // B. New Path (AmbiRotate - Folder Expansion)
        if (onFilesLoaded && window.electronAPI?.expandPaths) {
            const rawPaths = Array.from(e.dataTransfer.files).map(f => f.path);
            const expandedPaths = await window.electronAPI.expandPaths(rawPaths); // Returns flat list of valid audio paths
            
            const fileObjs = expandedPaths.map(p => ({
                path: p,
                name: p.split(/[/\\]/).pop() || 'unknown'
            }));
            onFilesLoaded(fileObjs);
        }
    };

    // 2. HANDLER: Click to Browse
    const handleClick = async () => {
        // Only trigger browse if we are using the new system
        if (onFilesLoaded && window.electronAPI?.selectFiles) {
            // Backend will handle the file dialog and folder expansion
            const paths = await window.electronAPI.selectFiles(); 
            if (!paths || paths.length === 0) return;

            const fileObjs = paths.map((p: string) => ({
                path: p,
                name: p.split(/[/\\]/).pop() || 'unknown'
            }));
            onFilesLoaded(fileObjs);
        }
    };

    return (
        <div 
            onDragOver={(e) => { e.preventDefault(); setIsDragOver(true); }}
            onDragLeave={() => setIsDragOver(false)}
            onDrop={handleDrop}
            onClick={handleClick}
            className={`
                ${className || ''} 
                flex-1 min-h-0 relative rounded-xl border-2 border-dashed transition-all cursor-pointer overflow-hidden flex flex-col
                ${isDragOver ? 'border-studio-accent bg-studio-accent/10' : 'border-gray-600 hover:border-gray-500 bg-studio-panel'}
            `}
        >
            <div className="flex-1 overflow-y-auto p-4 pointer-events-none">
                {children ? children : (
                    <div className="h-full flex flex-col items-center justify-center text-gray-400">
                        <svg className="w-12 h-12 mb-4 opacity-50" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M7 16a4 4 0 01-.88-7.903A5 5 0 1115.9 6L16 6a5 5 0 011 9.9M15 13l-3-3m0 0l-3 3m3-3v12" /></svg>
                        <p className="text-center font-medium">Drop files or folders here</p>
                        <p className="text-sm opacity-60 mt-1">(Click to browse)</p>
                    </div>
                )}
            </div>
        </div>
    );
};

```
