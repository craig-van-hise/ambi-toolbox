import AVFoundation
import AudioToolbox

func getAmbisonicLayout(channelCount: Int) -> Data? {
    // 1. Calculate the size of the layout plus the array of descriptions
    let layoutSize = MemoryLayout<AudioChannelLayout>.size + 
                     (MemoryLayout<AudioChannelDescription>.size * channelCount)
    
    // 2. Allocate the raw memory
    let pointer = UnsafeMutableRawPointer.allocate(byteCount: layoutSize, alignment: MemoryLayout<AudioChannelLayout>.alignment)
    
    // 3. Bind the header to AudioChannelLayout
    let layout = pointer.bindMemory(to: AudioChannelLayout.self, capacity: 1)
    
    // 4. Set the tag to kAudioChannelLayoutTag_UseChannelDescriptions (0)
    // This tells the system: "Ignore the tag, look at the individual channel labels."
    layout.pointee.mChannelLayoutTag = kAudioChannelLayoutTag_UseChannelDescriptions
    layout.pointee.mNumberChannelDescriptions = UInt32(channelCount)
    
    // 5. Fill each description with the HOA label
    // The constant for HOA ACN SN3D is kAudioChannelLabel_HOA_ACN_SN3D (200)
    let descriptionsBase = pointer.advanced(by: MemoryLayout<AudioChannelLayout>.size)
    let descriptions = descriptionsBase.bindMemory(to: AudioChannelDescription.self, capacity: channelCount)
    
    for i in 0..<channelCount {
        descriptions[i].mChannelLabel = kAudioChannelLabel_HOA_ACN_SN3D
        descriptions[i].mChannelFlags = []
        descriptions[i].mCoordinates = (0, 0, 0)
    }
    
    // 6. Wrap in Data for AVAssetWriter
    let data = Data(bytes: pointer, count: layoutSize)
    pointer.deallocate()
    
    return data
}
