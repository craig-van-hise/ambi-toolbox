import { dialog } from 'electron';
import fs from 'fs';
import path from 'path';

// Helper: Recursive File Scanner
const getAudioFiles = (dir: string, fileList: string[] = []) => {
    const files = fs.readdirSync(dir);
    for (const file of files) {
        const filePath = path.join(dir, file);
        const stat = fs.statSync(filePath);
        if (stat.isDirectory()) {
            getAudioFiles(filePath, fileList);
        } else {
            // Filter for Audio Extensions
            if (/\.(wav|opus|caf|flac|mp3|ogg|aac)$/i.test(file)) {
                fileList.push(filePath);
            }
        }
    }
    return fileList;
};

// 1. CLICK-TO-BROWSE HANDLER
ipcMain.handle('dialog:openFile', async () => {
    const { canceled, filePaths } = await dialog.showOpenDialog({
        properties: ['openFile', 'multiSelections', 'openDirectory'],
        filters: [{ name: 'Audio', extensions: ['wav', 'opus', 'caf', 'flac', 'mp3'] }]
    });
    if (canceled) return [];
    
    // Expand any selected folders immediately
    const allFiles: string[] = [];
    for (const p of filePaths) {
        if (fs.statSync(p).isDirectory()) {
            getAudioFiles(p, allFiles);
        } else {
            allFiles.push(p);
        }
    }
    return allFiles;
});

// 2. DROP-FOLDER EXPANDER
ipcMain.handle('app:expandPaths', async (event, paths: string[]) => {
    const allFiles: string[] = [];
    for (const p of paths) {
        try {
            if (fs.statSync(p).isDirectory()) {
                getAudioFiles(p, allFiles);
            } else if (/\.(wav|opus|caf|flac|mp3|ogg|aac)$/i.test(p)) {
                allFiles.push(p);
            }
        } catch (e) { console.error(`Skipping invalid path: ${p}`); }
    }
    return allFiles;
});

**Step 3: Create the Smart Drop Zone (`src/components/SmartDropZone.tsx`)**
status: ✅ Completed
Refactor the drag-and-drop logic into a reusable component that handles the folder expansion automatically.

```tsx
import React, { useState } from 'react';

interface Props {
    onFilesLoaded: (files: { path: string, name: string }[]) => void;
    children?: React.ReactNode;
}

export const SmartDropZone: React.FC<Props> = ({ onFilesLoaded, children }) => {
    const [isDragOver, setIsDragOver] = useState(false);

    const handleDrop = async (e: React.DragEvent) => {
        e.preventDefault();
        setIsDragOver(false);
        
        // 1. Get Raw Paths
        const rawPaths = Array.from(e.dataTransfer.files).map(f => f.path);
        
        // 2. Expand Folders via Backend
        const expandedPaths = await window.electronAPI.expandPaths(rawPaths);
        
        // 3. Convert to Objects
        const fileObjs = expandedPaths.map(p => ({
            path: p,
            name: p.split(/[/\\]/).pop() || 'unknown'
        }));
        
        onFilesLoaded(fileObjs);
    };

    const handleClick = async () => {
        const paths = await window.electronAPI.selectFiles();
        const fileObjs = paths.map((p: string) => ({
            path: p,
            name: p.split(/[/\\]/).pop() || 'unknown'
        }));
        if (fileObjs.length > 0) onFilesLoaded(fileObjs);
    };

    return (
        <div 
            onDragOver={(e) => { e.preventDefault(); setIsDragOver(true); }}
            onDragLeave={() => setIsDragOver(false)}
            onDrop={handleDrop}
            onClick={handleClick}
            className={`
                flex-1 min-h-0 relative rounded-xl border-2 border-dashed transition-all cursor-pointer overflow-hidden flex flex-col
                ${isDragOver ? 'border-studio-accent bg-studio-accent/10' : 'border-gray-600 hover:border-gray-500 bg-studio-panel'}
            `}
        >
            <div className="flex-1 overflow-y-auto p-4">
                {children ? children : (
                    <div className="h-full flex flex-col items-center justify-center text-gray-400 pointer-events-none">
                        <svg className="w-12 h-12 mb-4 opacity-50" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M7 16a4 4 0 01-.88-7.903A5 5 0 1115.9 6L16 6a5 5 0 011 9.9M15 13l-3-3m0 0l-3 3m3-3v12" /></svg>
                        <p className="text-center font-medium">Drop files or folders here</p>
                        <p className="text-sm opacity-60 mt-1">(Click to browse)</p>
                    </div>
                )}
            </div>
        </div>
    );
};

```
